% Two-line CV block: #1 left top, #2 right top, #3 left bottom, #4 right bottom (bold / italic).
\newcommand{\cvTwoLineBlock}[4]{%
    \noindent
    \begin{minipage}[t]{0.48\linewidth}\raggedright{\large\bfseries #1}\end{minipage}%
    \hfill
    \begin{minipage}[t]{0.48\linewidth}\raggedleft\itshape #2\end{minipage}\\[0.2em]
    \noindent
    \begin{minipage}[t]{0.48\linewidth}\raggedright\bfseries #3\end{minipage}%
    \hfill
    \begin{minipage}[t]{0.48\linewidth}\raggedleft\itshape #4\end{minipage}%
    \par\vspace{0.45em}%
}
% Jobs: \addvspace separates entries. #1 company, #2 location, #3 role, #4 dates.
\newcommand{\cvExperienceHeader}[4]{%
    \par\addvspace{0.95\baselineskip}%
    \cvTwoLineBlock{#1}{#2}{#3}{#4}%
}
% Education: university | location, then degree | dates (slightly tighter vertical gap than jobs).
\newcommand{\cvEducationHeader}[4]{%
    \par\addvspace{0.65\baselineskip}%
    \cvTwoLineBlock{#1}{#2}{#3}{#4}%
}

% Experience entry wrapper so `content.tex` can define a list without One/Two/Three variables.
% Usage: \Experience{company}{location}{role}{period}{\item ...}{skills line}
\newcommand{\Experience}[6]{%
    \cvExperienceHeader{#1}{#2}{#3}{#4}%
    \begin{itemize}
        #5
    \end{itemize}
    #6
}

% Education entry wrapper.
% Usage: \Education{university}{location}{degree}{period}{\item ...}{skills line}
\newcommand{\Education}[6]{%
    \cvEducationHeader{#1}{#2}{#3}{#4}%
    \begin{itemize}
        #5
    \end{itemize}
    #6
}

% Patent entry wrapper.
% Usage: \Patent{title}{inventors}{year}{type}{id}{link}
\newcommand{\Patent}[6]{%
    \subsection{#1}%
    \textit{#2} \hfill \textit{#3}%
    \begin{itemize}
        \item \textit{#4} - #5%
        \item \href{#6}{#5}%
    \end{itemize}
}

% Publication entry wrapper.
% Usage: \Publication{title}{authors}{year}{type}{doi}
\newcommand{\Publication}[5]{%
    \subsection{#1}%
    \textit{#2} \hfill \textit{#3}%
    \begin{itemize}
        \item \textit{#4} - DOI: \href{https://doi.org/#5}{#5}%
    \end{itemize}
}

% Language entry wrapper.
% Usage: \Language{language}{proficiency}
\newcommand{\Language}[2]{%
    \item \textbf{#1}: #2%
}
