% -----------------------------------------------------------------------------
% Entry headers (single-column / ATS-friendly reading order)
% Args: {organization}{location}{role or degree}{period}
% -----------------------------------------------------------------------------
\newcommand{\cvEntryHeader}[4]{%
    \noindent{\large\bfseries #3}\\[0.12em]%
    \noindent #1\ \textbar\ #2\ \textbar\ \textit{#4}%
    \par\vspace{0.45em}%
}

\newcommand{\cvExperienceHeader}[4]{%
    \par\addvspace{0.95\baselineskip}%
    \cvEntryHeader{#1}{#2}{#3}{#4}%
}

\newcommand{\cvEducationHeader}[4]{%
    \par\addvspace{0.65\baselineskip}%
    \cvEntryHeader{#1}{#2}{#3}{#4}%
}

% \Experience{company}{location}{role}{period}{bullet items}{skills line}
\newcommand{\Experience}[6]{%
    \cvExperienceHeader{#1}{#2}{#3}{#4}%
    \begin{itemize}
        #5
    \end{itemize}
    #6
}

% \Education{university}{location}{degree}{period}{bullet items}{skills line}
% Leave bullet items empty ({}) if the degree line is enough.
\newcommand{\Education}[6]{%
    \cvEducationHeader{#1}{#2}{#3}{#4}%
    \if\relax\detokenize{#5}\relax\else
        \begin{itemize}
            #5
        \end{itemize}
    \fi
    #6
}

% \Patent{title}{inventors}{year}{type}{id}{url}
\newcommand{\Patent}[6]{%
    \par\addvspace{0.85\baselineskip}%
    \noindent{\large\bfseries #1}\par\vspace{0.15em}%
    \noindent\textit{#2}\\[0.1em]
    \noindent\textit{#3}%
    \begin{itemize}
        \item \textit{#4} --- #5%
        \item \href{#6}{#5}%
    \end{itemize}
}

% \Publication{title}{authors}{year}{type}{doi}
\newcommand{\Publication}[5]{%
    \par\addvspace{0.85\baselineskip}%
    \noindent{\large\bfseries #1}\par\vspace{0.15em}%
    \noindent\textit{#2}\\[0.1em]
    \noindent\textit{#3}%
    \begin{itemize}
        \item \textit{#4} --- DOI: \href{https://doi.org/#5}{#5}%
    \end{itemize}
}

% \Language{language}{proficiency}
\newcommand{\Language}[2]{%
