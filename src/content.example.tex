% --- Professional experience -------------------------------------------------
% \Experience{company}{location}{role}{period}{items}{skills}

    \Experience
        {Company Name}
        {City, Country}
        {Job Title}
        {Start Date -- End Date}
        {
            \item Achievement or responsibility 1
            \item Achievement or responsibility 2
            \item Achievement or responsibility 3
        }
        {
            {\small\textit{Skills:} Skill 1, Skill 2, Skill 3, Skill 4}
        }

    \Experience
        {Another Company}
        {City, Country}
        {Job Title 2}
        {Start Date -- End Date}
        {
            \item Achievement or responsibility 1
            \item Achievement or responsibility 2
        }
        {
            {\small\textit{Skills:} Skill 1, Skill 2}
        }
}

% --- Patents -----------------------------------------------------------------
