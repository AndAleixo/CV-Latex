% Two-line CV block: #1 left top, #2 right top, #3 left bottom, #4 right bottom (bold / italic).
\newcommand{\cvTwoLineBlock}[4]{%
    \noindent
    \begin{minipage}[t]{0.48\linewidth}\raggedright{\large\bfseries #1}\end{minipage}%
    \hfill
    \begin{minipage}[t]{0.48\linewidth}\raggedleft\itshape #2\end{minipage}\\[0.2em]
    \noindent
    \begin{minipage}[t]{0.48\linewidth}\raggedright\bfseries #3\end{minipage}%
    \hfill
    \begin{minipage}[t]{0.48\linewidth}\raggedleft\itshape #4\end{minipage}%
    \par\vspace{0.45em}%
}
% Jobs: \addvspace separates entries. #1 company, #2 location, #3 role, #4 dates.
\newcommand{\cvExperienceHeader}[4]{%
    \par\addvspace{0.95\baselineskip}%
    \cvTwoLineBlock{#1}{#2}{#3}{#4}%
}
% Education: university | location, then degree | dates (slightly tighter vertical gap than jobs).
\newcommand{\cvEducationHeader}[4]{%
    \par\addvspace{0.65\baselineskip}%
    \cvTwoLineBlock{#1}{#2}{#3}{#4}%
}
